# Q8: Flatten a Nested Dictinary.

data = {'a':{'b':{'c':42},'d':7},'e':10}

def flatten_dict(data, parent_key='',sep='.'):
    items = {}
    for key, value in data.items():
        new_keys = f"{parent_key}{sep}{key}" if parent_key else key
        if isinstance(value, dict):
            items.update(flatten_dict(value, new_keys, sep))
        else:
            items[new_keys] = value
    return
print(flatten_dict(data))


# Q:9 sorted dictionary by values.

data = {'a':5,'b':9,'c':2,'d':7}

def sort_by_values(data):
    sorted_items = sorted(data.items(),
                          key = lambda item: item[1])
    return{key: value for key, value in sorted_items}
print(sort_by_values(data))
